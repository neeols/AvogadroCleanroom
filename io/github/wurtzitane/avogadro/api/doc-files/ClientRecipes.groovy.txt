        return AvogadroApi.builder("guide:pure_water").name("Purified water").smiles("O")

    // BEGIN stack-matching

    // BEGIN remap
    public static StructureEditResult isolateOxygen() {
        // Both selections refer to the ORIGINAL three-atom water fragment.
        // Deletion resolves the whole selection before removing any atoms.
        return GuideModels.water().edit().removeAtoms(1, 2).build();
        // oldToNew() is [0, -1, -1]. The resulting O atom is not a water molecule.
    }
    // END remap

    // BEGIN saved-edits
    public static StructureEdits hydrogenPeroxideEdits() {
        // Water indices: O=0, H=1, H=2. Replace H=2 with O, then add its H.
        return StructureEdits.builder().replaceAtom(2, "O").moveAtom(2, -.9, -.95, 0)
                .atom("H", -1.65, -.6, .4).setBond(2, 3, BondType.SINGLE).build();
    }
    // END saved-edits

    // BEGIN definition-edit
    public static MoleculeDef peroxide(MoleculeDef water) {
        // Carries over names and associations. Rename/rebind deliberately if needed.
        return MoleculeEdits.apply("guide:peroxide", water, hydrogenPeroxideEdits(), false);
    }
    // END definition-edit

    // BEGIN copy-edit
    public static MoleculeSpec peroxideCopy() {
        return AvogadroApi.builder("guide:peroxide").copy("guide:water")
                .name("Hydrogen peroxide").edits(hydrogenPeroxideEdits()).build();
    }
    // END copy-edit

    // BEGIN transform
    public static StructureFragment waterPair() {
        def water = GuideModels.water();
        // A disconnected pair of water molecules, not an inferred hydrogen bond.
        return water.qualified("first").append("second",
                water.rotate(0, 0, 1, 180).translate(3.2, 0, 0));
    }
    // END transform

    // BEGIN crown-ring
    public static StructureAssembly crownRing() {
        def unit = new StructureBuilder();
        for (int i = 0; i < 3; i++) {
            double angle = Math.toRadians(i * 20);
            double x = 4.4 * Math.cos(angle), y = 4.4 * Math.sin(angle);
            int atom = unit.addAtom(i == 0 ? "O" : "C", x, y, 0);
            if (i > 0) {
                int h1 = unit.addAtom("H", x, y, .95);
                int h2 = unit.addAtom("H", x, y, -.95);
                unit.bond(atom, h1).bond(atom, h2);
            }
        }
        // Indices after H insertion: O=0, first C=1, second C=4.
        def fragment = unit.bond(0, 1).bond(1, 4).attachment("head", 0).attachment("tail", 4).build();
        double c = Math.cos(Math.PI / 3), s = Math.sin(Math.PI / 3);
        double[][] turn = [[c,-s,0,0], [s,c,0,0], [0,0,1,0]] as double[][];
        return StructureAssembly.builder()
                .repeat("ring", StructureAssembly.builder().part("unit", fragment).build(), 6, turn)
                .neighbor("unit/tail", "unit/head", 1, BondType.SINGLE)
                // Neighbor links leave the boundary open; this closes it explicitly.
                .link("ring[last]/unit/tail", "ring[first]/unit/head", BondType.SINGLE).build();
    }
    // END crown-ring

    // BEGIN random-links
    public static PolymerNetworkDefinition reactiveSites() {
        // Two authored hydroxyl radicals. At p=1 they form H–O–O–H.
        // At p=0 they remain radicals, not two water molecules.
        def left = new StructureBuilder().atom("O", -.7, 0, 0).atom("H", -1.1, .85, 0)
                .bond(0, 1).attachment("site", 0).attachment("alias", 0).randomLinkCapacity(0, 1).build();
        def right = new StructureBuilder().atom("O", .7, 0, 0).atom("H", 1.1, -.85, 0)
                .bond(0, 1).attachment("site", 0).randomLinkCapacity(0, 1).build();
        def assembly = StructureAssembly.builder().part("left", left).part("right", right)
                .randomLinks("peroxide_bond", "left/site", "right/site", "paired", "chance", BondType.SINGLE)
                .build();
        return new PolymerNetworkDefinition(List.of(
                PolymerNetworkDefinition.Parameter.probability("chance", 1)), assembly, 42L);
    }
    // END random-links

    // BEGIN listeners
    public static void watch(ExportJob job, java.util.function.Consumer<String> showStatus) {
        job.onComplete(result -> {
            // This listener runs on the client thread after capture/encoder cleanup.
            switch (result.state()) {
                case SUCCEEDED -> showStatus.accept("Saved " + result.output());
                case CANCELLED -> showStatus.accept("Export cancelled");
                case FAILED -> showStatus.accept("Export failed: " + result.failure());
                default -> throw new IllegalStateException("Expected a terminal result");
            }
        });
    }
    // END listeners

    // BEGIN progress
    public static String progress(ExportJob job) {
        def progress = job.progress(); // immutable, safe to read from another thread
        return progress.state() + ": " + progress.encodedFrames() + "/" + progress.totalFrames();
    }
    // END progress

    // BEGIN cancel
    public static boolean cancel(ExportJob job) {
        // Requests cancellation; result() remains null until cleanup finishes.
        return job.cancel();
    }
    // END cancel

    // BEGIN client-thread
    public static void exportFromWorker(MoleculeDef molecule, java.util.function.Consumer<ExportJob> ready) {
        Minecraft.getMinecraft().addScheduledTask(() -> {
            try {
                ready.accept(exportGif(molecule));
            } catch (IOException failure) {
                throw new IllegalStateException("Could not start export", failure);
            }
        });
    }
    // END client-thread
